```json

  "message": "string (Markdown formatted text for rendering)",
  "citations": [
    {
      "policy_name": "string",

      "content_snippet": "string",

    }
  ],
  "suggested_action": {
    "tool_name": "submit_leave_request",
    "parameters": {
      "user_id": "string",

}
```
